cmake_minimum_required(VERSION 3.16)
project(CliqueTwin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(ct_edge_query examples/edge_query.cpp)
add_executable(ct_triangle_online applications/triangle/triangle_ct_online.cpp)
add_executable(ct_structure experiments/ct_structure.cpp)
add_executable(ct_query_benchmark experiments/ct_query_benchmark.cpp)

if (MSVC)
  foreach(t ct_edge_query ct_triangle_online ct_structure ct_query_benchmark)
    target_compile_options(${t} PRIVATE /O2)
  endforeach()
else()
  foreach(t ct_edge_query ct_triangle_online ct_structure ct_query_benchmark)
    target_compile_options(${t} PRIVATE -O3 -Wall -Wextra)
  endforeach()
endif()
